  <script>
    const STORAGE_KEY = "smartspend-transactions";
    const BUDGET_KEY = "smartspend-monthly-budget";

    const categories = {
      expense: ["Food", "Transport", "Shopping", "Education", "Bills", "Health", "Entertainment", "Other"],
      income: ["Pocket money", "Salary", "Scholarship", "Gift", "Other"]
    };

    const currency = new Intl.NumberFormat("en-IN", {
      style: "currency",
      currency: "INR",
      maximumFractionDigits: 2
    });

    const monthPicker = document.getElementById("monthPicker");
    const transactionRows = document.getElementById("transactionRows");
    const modalBackdrop = document.getElementById("modalBackdrop");
    const transactionForm = document.getElementById("transactionForm");
    const categorySelect = document.getElementById("transactionCategory");

    let transactions = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
    let monthlyBudget = Number(localStorage.getItem(BUDGET_KEY)) || 5000;
    let editingId = null;
    let categoryChart = null;
    let trendChart = null;

    function localDateString(date = new Date()) {
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, "0");
      const day = String(date.getDate()).padStart(2, "0");
      return `${year}-${month}-${day}`;
    }

    function currentMonthString() {
      return localDateString().slice(0, 7);
    }

    function money(amount) {
      return currency.format(amount);
    }

    function escapeHtml(value) {
      return String(value).replace(/[&<>"']/g, character => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;"
      })[character]);
    }

    function saveTransactions() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(transactions));
    }

    function monthTransactions(month) {
      return transactions.filter(transaction => transaction.date.startsWith(month));
    }

    function setCategories(type, selectedCategory = "") {
      categorySelect.innerHTML = categories[type]
        .map(category => `<option value="${escapeHtml(category)}">${escapeHtml(category)}</option>`)
        .join("");

      if (categories[type].includes(selectedCategory)) {
        categorySelect.value = selectedCategory;
      }
    }

    function renderDashboard() {
      const month = monthPicker.value || currentMonthString();
      const records = monthTransactions(month);
      const income = records
        .filter(item => item.type === "income")
        .reduce((sum, item) => sum + Number(item.amount), 0);
      const expenses = records
        .filter(item => item.type === "expense")
        .reduce((sum, item) => sum + Number(item.amount), 0);
      const balance = income - expenses;
      const savings = income > 0 ? Math.round((balance / income) * 100) : 0;

      document.getElementById("incomeTotal").textContent = money(income);
      document.getElementById("expenseTotal").textContent = money(expenses);

      const balanceElement = document.getElementById("balanceTotal");
      balanceElement.textContent = money(balance);
      balanceElement.className = `summary-value ${balance < 0 ? "negative" : "positive"}`;

      const savingsElement = document.getElementById("savingsRate");
      savingsElement.textContent = `${savings}%`;
      savingsElement.className = `summary-value ${savings < 0 ? "negative" : "positive"}`;

      renderBudget(expenses);
      renderCharts(month, records);
      renderTransactions();
    }

    function renderBudget(expenses) {
      const percent = monthlyBudget > 0 ? (expenses / monthlyBudget) * 100 : 0;
      const progress = document.getElementById("budgetProgress");

      document.getElementById("budgetInput").value = monthlyBudget;
      document.getElementById("budgetSpent").textContent = `${money(expenses)} spent`;
      document.getElementById("budgetRemaining").textContent =
        `${money(Math.max(monthlyBudget - expenses, 0))} remaining`;

      progress.style.width = `${Math.min(percent, 100)}%`;
      progress.className = "budget-progress-fill";

      const message = document.getElementById("budgetMessage");
      if (percent >= 100) {
        progress.classList.add("over");
        message.textContent = `Budget exceeded by ${money(expenses - monthlyBudget)}.`;
        message.className = "budget-message negative";
      } else if (percent >= 80) {
        progress.classList.add("warning");
        message.textContent = `Heads up — you have used ${Math.round(percent)}% of your budget.`;
        message.className = "budget-message";
        message.style.color = "var(--yellow)";
      } else {
        message.textContent = "You're within your budget. Keep it up!";
        message.className = "budget-message positive";
        message.style.color = "";
      }
    }

    function renderTransactions() {
      const month = monthPicker.value || currentMonthString();
      const query = document.getElementById("searchInput").value.trim().toLowerCase();
      const type = document.getElementById("typeFilter").value;

      const filtered = monthTransactions(month)
        .filter(item => type === "all" || item.type === type)
        .filter(item => `${item.description} ${item.category}`.toLowerCase().includes(query))
        .sort((a, b) => b.date.localeCompare(a.date));

      transactionRows.innerHTML = filtered.map(item => `
        <tr>
          <td>${escapeHtml(new Date(`${item.date}T00:00:00`).toLocaleDateString("en-IN", {
            day: "numeric", month: "short", year: "numeric"
          }))}</td>
          <td>${escapeHtml(item.description)}</td>
          <td><span class="category-pill">${escapeHtml(item.category)}</span></td>
          <td>${item.type === "income" ? "Income" : "Expense"}</td>
          <td class="${item.type === "income" ? "positive" : "negative"}">
            ${item.type === "income" ? "+" : "−"}${money(Number(item.amount))}
          </td>
          <td>
            <div class="row-actions">
              <button type="button" data-action="edit" data-id="${escapeHtml(item.id)}">Edit</button>
              <button type="button" data-action="delete" data-id="${escapeHtml(item.id)}">Delete</button>
            </div>
          </td>
        </tr>
      `).join("");

      document.getElementById("emptyState").hidden = filtered.length !== 0;
    }

    function renderCharts(month, records) {
      if (!window.Chart) {
        document.getElementById("categoryChart").hidden = true;
        document.getElementById("trendChart").hidden = true;
        document.getElementById("categoryChartFallback").style.display = "block";
        document.getElementById("trendChartFallback").style.display = "block";
        document.getElementById("categoryChartFallback").textContent =
          "Charts require an internet connection to load Chart.js.";
        document.getElementById("trendChartFallback").textContent =
          "Charts require an internet connection to load Chart.js.";
        return;
      }

      const categoryTotals = {};
      records.filter(item => item.type === "expense").forEach(item => {
        categoryTotals[item.category] = (categoryTotals[item.category] || 0) + Number(item.amount);
      });

      const palette = ["#7464e8", "#20a779", "#f0a33a", "#e36775", "#4299d5", "#a06ad8", "#58b5ad", "#9aa2b4"];
      const categoryLabels = Object.keys(categoryTotals);

      if (categoryChart) categoryChart.destroy();
      categoryChart = new Chart(document.getElementById("categoryChart"), {
        type: "doughnut",
        data: {
          labels: categoryLabels.length ? categoryLabels : ["No expenses yet"],
          datasets: [{
            data: categoryLabels.length ? Object.values(categoryTotals) : [1],
            backgroundColor: categoryLabels.length ? palette.slice(0, categoryLabels.length) : ["#eceef4"],
            borderWidth: 0
          }]
        },
        options: {
          maintainAspectRatio: false,
          cutout: "68%",
          plugins: {
            legend: { position: "bottom", labels: { usePointStyle: true, padding: 17 } },
            tooltip: {
              callbacks: {
                label: context => categoryLabels.length
                  ? `${context.label}: ${money(context.raw)}`
                  : "Add expenses to see your spending"
              }
            }
          }
        }
      });

      const [year, monthNumber] = month.split("-").map(Number);
      const labels = [];
      const incomeData = [];
      const expenseData = [];

      for (let offset = 5; offset >= 0; offset--) {
        const date = new Date(year, monthNumber - 1 - offset, 1);
        const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
        const monthRecords = monthTransactions(key);

        labels.push(date.toLocaleDateString("en-IN", { month: "short" }));
        incomeData.push(monthRecords
          .filter(item => item.type === "income")
          .reduce((sum, item) => sum + Number(item.amount), 0));
        expenseData.push(monthRecords
          .filter(item => item.type === "expense")
          .reduce((sum, item) => sum + Number(item.amount), 0));
      }

      if (trendChart) trendChart.destroy();
      trendChart = new Chart(document.getElementById("trendChart"), {
        type: "bar",
        data: {
          labels,
          datasets: [
            {
              label: "Income",
              data: incomeData,
              backgroundColor: "#20a779",
              borderRadius: 5
            },
            {
              label: "Expenses",
              data: expenseData,
              backgroundColor: "#7464e8",
              borderRadius: 5
            }
          ]
        },
        options: {
          maintainAspectRatio: false,
          scales: {
            y: {
              beginAtZero: true,
              ticks: { callback: value => `₹${Number(value).toLocaleString("en-IN")}` },
              grid: { color: "#f0f1f5" }
            },
            x: { grid: { display: false } }
          },
          plugins: {
            legend: { position: "bottom", labels: { usePointStyle: true, padding: 17 } },
            tooltip: {
              callbacks: { label: context => `${context.dataset.label}: ${money(context.raw)}` }
            }
          }
        }
      });
    }

    function openModal(transaction = null) {
      editingId = transaction ? transaction.id : null;
      transactionForm.reset();
      document.getElementById("modalTitle").textContent = transaction ? "Edit transaction" : "Add transaction";
      document.getElementById("transactionType").value = transaction ? transaction.type : "expense";
      setCategories(
        document.getElementById("transactionType").value,
        transaction ? transaction.category : ""
      );
      document.getElementById("transactionAmount").value = transaction ? transaction.amount : "";
      document.getElementById("transactionDate").value = transaction ? transaction.date : localDateString();
      document.getElementById("transactionDescription").value = transaction ? transaction.description : "";
      modalBackdrop.classList.add("open");
      document.getElementById("transactionAmount").focus();
    }

    function closeModal() {
      modalBackdrop.classList.remove("open");
      editingId = null;
    }

    function exportCsv() {
      if (transactions.length === 0) {
        alert("There are no transactions to export yet.");
        return;
      }

      const headers = ["Date", "Description", "Category", "Type", "Amount"];
      const rows = transactions
        .slice()
        .sort((a, b) => a.date.localeCompare(b.date))
        .map(item => [item.date, item.description, item.category, item.type, item.amount]);

      const csv = [headers, ...rows]
        .map(row => row.map(value => `"${String(value).replace(/"/g, '""')}"`).join(","))
        .join("\r\n");

      const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = `smartspend-transactions-${currentMonthString()}.csv`;
      link.click();
      URL.revokeObjectURL(url);
    }

    });

      if (event.key === "Escape") closeModal();
    });

      setCategories(event.target.value);
    });

    transactionForm.addEventListener("submit", event => {
      event.preventDefault();

      const transaction = {
        id: editingId || (crypto.randomUUID ? crypto.randomUUID() : String(Date.now())),
        type: document.getElementById("transactionType").value,
        amount: Number(document.getElementById("transactionAmount").value),
        category: categorySelect.value,
        date: document.getElementById("transactionDate").value,
        description: document.getElementById("transactionDescription").value.trim()
      };

      if (!transaction.amount || transaction.amount <= 0 || !transaction.date || !transaction.description) {
        alert("Please enter a valid amount, date, and description.");
        return;
      }

      if (editingId) {
        transactions = transactions.map(item => item.id === editingId ? transaction : item);
      } else {
        transactions.push(transaction);
      }

      saveTransactions();
      closeModal();
      renderDashboard();
    });

    transactionRows.addEventListener("click", event => {
      const button = event.target.closest("button[data-action]");
      if (!button) return;

      const transaction = transactions.find(item => item.id === button.dataset.id);
      if (!transaction) return;

      if (button.dataset.action === "edit") {
        openModal(transaction);
      } else if (button.dataset.action === "delete") {
        if (confirm(`Delete "${transaction.description}"?`)) {
          transactions = transactions.filter(item => item.id !== transaction.id);
          saveTransactions();
          renderDashboard();
        }
      }
    });

    document.getElementById("budgetForm").addEventListener("submit", event => {
      event.preventDefault();
      const newBudget = Number(document.getElementById("budgetInput").value);

      if (!Number.isFinite(newBudget) || newBudget <= 0) {
        alert("Please enter a budget greater than zero.");
        return;
      }

      monthlyBudget = newBudget;
      localStorage.setItem(BUDGET_KEY, String(monthlyBudget));
      renderDashboard();
    });
